def bfs(g, src):                                   # shortest path in edges
    dist = {src: 0}; q = deque([src])
    while q:
        u = q.popleft()
        for v in g[u]:
            if v not in dist:
                dist[v] = dist[u] + 1; q.append(v)
    return dist

def num_islands(grid):
    R, C = len(grid), len(grid[0]); count = 0
    def sink(r, c):                                # iterative DFS, marks in place
        st = [(r, c)]; grid[r][c] = '0'
        while st:
            r, c = st.pop()
            for nr, nc in ((r+1,c), (r-1,c), (r,c+1), (r,c-1)):
                if 0 <= nr < R and 0 <= nc < C and grid[nr][nc] == '1':
                    grid[nr][nc] = '0'; st.append((nr, nc))
    for r in range(R):
        for c in range(C):
            if grid[r][c] == '1': count += 1; sink(r, c)
    return count
# max area of island: return size from sink; surrounded regions: flood from border first

def rotting_oranges(grid):                         # multi-source BFS: all sources start at distance 0
    R, C = len(grid), len(grid[0]); q = deque(); fresh = 0
    for r in range(R):
        for c in range(C):
            if grid[r][c] == 2: q.append((r, c, 0))
            elif grid[r][c] == 1: fresh += 1
    t = 0
    while q:
        r, c, t = q.popleft()
        for nr, nc in ((r+1,c), (r-1,c), (r,c+1), (r,c-1)):
            if 0 <= nr < R and 0 <= nc < C and grid[nr][nc] == 1:
                grid[nr][nc] = 2; fresh -= 1; q.append((nr, nc, t + 1))
    return -1 if fresh else t

def zero_one_bfs(g, src, n):                       # edge weights 0 or 1: deque, push-front on 0
    dist = [INF] * n; dist[src] = 0; dq = deque([src])
    while dq:
        u = dq.popleft()
        for v, w in g[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                dq.appendleft(v) if w == 0 else dq.append(v)
    return dist

def is_bipartite(g, n):
    color = [-1] * n
    for s in range(n):
        if color[s] != -1: continue
        color[s] = 0; q = deque([s])
        while q:
            u = q.popleft()
            for v in g[u]:
                if color[v] == -1: color[v] = color[u] ^ 1; q.append(v)
                elif color[v] == color[u]: return False
    return True

def clone_graph(node):
    if not node: return None
    copy = {node: Node(node.val)}; q = deque([node])
    while q:
        u = q.popleft()
        for v in u.neighbors:
            if v not in copy: copy[v] = Node(v.val); q.append(v)
            copy[u].neighbors.append(copy[v])
    return copy[node]

def word_ladder(begin, end, words):                # BFS over words, neighbors via wildcard buckets
    words = set(words)
    if end not in words: return 0
    L = len(begin); buckets = defaultdict(list)
    for w in words:
        for i in range(L): buckets[w[:i] + '*' + w[i+1:]].append(w)
    q = deque([(begin, 1)]); seen = {begin}
    while q:
        w, d = q.popleft()
        if w == end: return d
        for i in range(L):
            for nxt in buckets[w[:i] + '*' + w[i+1:]]:
                if nxt not in seen: seen.add(nxt); q.append((nxt, d + 1))
    return 0

def shortest_path_binary_matrix(grid):             # 8-directional BFS, return path length in cells
    n = len(grid)
    if grid[0][0] or grid[n-1][n-1]: return -1
    q = deque([(0, 0, 1)]); grid[0][0] = 1
    while q:
        r, c, d = q.popleft()
        if r == c == n - 1: return d
        for dr in (-1, 0, 1):
            for dc in (-1, 0, 1):
                nr, nc = r + dr, c + dc
                if 0 <= nr < n and 0 <= nc < n and grid[nr][nc] == 0:
                    grid[nr][nc] = 1; q.append((nr, nc, d + 1))
    return -1

def course_schedule_ii(n, edges):                  # Kahn's on prerequisite pairs; returns [] if cycle. Full topo card in DS templates.
    g = defaultdict(list); indeg = [0] * n
    for a, b in edges: g[b].append(a); indeg[a] += 1     # b before a

        u = q.popleft(); order.append(u)

    return order if len(order) == n else []

def has_cycle_directed(n, g):                      # colors: 0 white, 1 gray (on stack), 2 black
    color = [0] * n
    def dfs(u):
        color[u] = 1
        for v in g[u]:
            if color[v] == 1: return True
            if color[v] == 0 and dfs(v): return True
        color[u] = 2; return False
    return any(color[u] == 0 and dfs(u) for u in range(n))

def has_cycle_undirected(n, g):                    # skip the edge back to parent; or DSU union returning False
    seen = [False] * n
    def dfs(u, parent):
        seen[u] = True
        for v in g[u]:
            if not seen[v]:
                if dfs(v, u): return True
            elif v != parent: return True
        return False
    return any(not seen[u] and dfs(u, -1) for u in range(n))

def dijkstra(g, src, n):                           # non-negative weights, lazy deletion
    dist = [INF] * n; dist[src] = 0; h = [(0, src)]
    while h:
        d, u = heapq.heappop(h)
        if d > dist[u]: continue                   # stale entry
        for v, w in g[u]:
            nd = d + w
            if nd < dist[v]:
                dist[v] = nd; heapq.heappush(h, (nd, v))
    return dist

def bellman_ford(n, edges, src):                   # handles negatives, detects negative cycle
    dist = [INF] * n; dist[src] = 0
    for _ in range(n - 1):
        changed = False
        for u, v, w in edges:
            if dist[u] + w < dist[v]: dist[v] = dist[u] + w; changed = True
        if not changed: break
    if any(dist[u] + w < dist[v] for u, v, w in edges): return None   # negative cycle
    return dist

def floyd(n, edges):                               # all pairs, O(n^3), n <= ~300
    d = [[INF] * n for _ in range(n)]
    for i in range(n): d[i][i] = 0
    for u, v, w in edges: d[u][v] = min(d[u][v], w)
    for k in range(n):
        for i in range(n):
            if d[i][k] == INF: continue
            for j in range(n):
                if d[i][k] + d[k][j] < d[i][j]: d[i][j] = d[i][k] + d[k][j]
    return d

def kruskal(n, edges):                             # edges (w, u, v)
    dsu = DSU(n); total = 0
    for w, u, v in sorted(edges):
        if dsu.union(u, v): total += w
    return total
def prim(g, n):                                    # g[u] = [(w, v)]
    seen = [False] * n; h = [(0, 0)]; total = 0
    while h:
        w, u = heapq.heappop(h)
        if seen[u]: continue
        seen[u] = True; total += w
        for nw, v in g[u]:
            if not seen[v]: heapq.heappush(h, (nw, v))
    return total

def reconstruct_itinerary(tickets):                # Eulerian path, Hierholzer, lexical order
    g = defaultdict(list)
    for a, b in sorted(tickets, reverse=True): g[a].append(b)   # pop() gives smallest
    route = []
    def dfs(u):
        while g[u]: dfs(g[u].pop())
        route.append(u)
    dfs('JFK'); return route[::-1]

def evaluate_division(equations, values, queries): # weighted graph, DFS/BFS for product along the path
    g = defaultdict(dict)
    for (a, b), v in zip(equations, values): g[a][b] = v; g[b][a] = 1 / v
    def solve(a, b):
        if a not in g or b not in g: return -1.0
        st = [(a, 1.0)]; seen = {a}
        while st:
            u, acc = st.pop()
            if u == b: return acc
            for v, w in g[u].items():
                if v not in seen: seen.add(v); st.append((v, acc * w))
        return -1.0
    return [solve(a, b) for a, b in queries]

def tarjan_bridges(n, g):                          # Critical Connections
    disc = [-1] * n; low = [0] * n; t = 0; out = []
    def dfs(u, p):
        nonlocal t
        disc[u] = low[u] = t; t += 1
        for v in g[u]:
            if v == p: continue
            if disc[v] == -1:
                dfs(v, u); low[u] = min(low[u], low[v])
                if low[v] > disc[u]: out.append((u, v))
            else: low[u] = min(low[u], disc[v])
    for u in range(n):
        if disc[u] == -1: dfs(u, -1)
    return out

def permutations_(nums):
    out, path, used = [], [], [False] * len(nums)
    def bt():
        if len(path) == len(nums): out.append(path[:]); return
        for i, x in enumerate(nums):
            if used[i]: continue
            # duplicates: if i and nums[i] == nums[i-1] and not used[i-1]: continue  (sorted input)
            used[i] = True; path.append(x); bt(); path.pop(); used[i] = False
    bt(); return out

def combination_sum(cands, target):                # unlimited reuse
    cands.sort(); out, path = [], []
    def bt(start, rem):
        if rem == 0: out.append(path[:]); return
        for i in range(start, len(cands)):
            if cands[i] > rem: break               # prune
            path.append(cands[i]); bt(i, rem - cands[i]); path.pop()   # i, not i+1: reuse allowed
    bt(0, target); return out

def generate_parentheses(n):
    out = []
    def bt(s, open_, close):
        if len(s) == 2 * n: out.append(s); return
        if open_ < n: bt(s + '(', open_ + 1, close)
        if close < open_: bt(s + ')', open_, close + 1)
    bt('', 0, 0); return out

def n_queens(n):
    out = []; cols, d1, d2 = set(), set(), set(); board = []
    def bt(r):
        if r == n: out.append(['.' * c + 'Q' + '.' * (n - c - 1) for c in board]); return
        for c in range(n):
            if c in cols or r - c in d1 or r + c in d2: continue
            cols.add(c); d1.add(r - c); d2.add(r + c); board.append(c)
            bt(r + 1)
            cols.discard(c); d1.discard(r - c); d2.discard(r + c); board.pop()
    bt(0); return out

def exist(board, word):                            # Word Search
    R, C = len(board), len(board[0])
    def dfs(r, c, i):
        if i == len(word): return True
        if not (0 <= r < R and 0 <= c < C) or board[r][c] != word[i]: return False
        tmp, board[r][c] = board[r][c], '#'        # mark visited in place
        found = any(dfs(r+dr, c+dc, i+1) for dr, dc in ((1,0),(-1,0),(0,1),(0,-1)))
        board[r][c] = tmp
        return found
    return any(dfs(r, c, 0) for r in range(R) for c in range(C))

def palindrome_partition(s):
    out, path = [], []
    def bt(start):
        if start == len(s): out.append(path[:]); return
        for end in range(start + 1, len(s) + 1):
            piece = s[start:end]
            if piece == piece[::-1]:
                path.append(piece); bt(end); path.pop()
    bt(0); return out

def restore_ip(s):
    out = []
    def bt(start, parts):
        if len(parts) == 4:
            if start == len(s): out.append('.'.join(parts))

        for L in (1, 2, 3):
            seg = s[start:start + L]

            bt(start + L, parts + [seg])
    bt(0, []); return out

def solve_sudoku(board):
    rows = [set() for _ in range(9)]; cols = [set() for _ in range(9)]; boxes = [set() for _ in range(9)]; empties = []
    for r in range(9):
        for c in range(9):
            v = board[r][c]
            if v == '.': empties.append((r, c))
            else: rows[r].add(v); cols[c].add(v); boxes[r//3*3 + c//3].add(v)
    def bt(i):
        if i == len(empties): return True
        r, c = empties[i]; b = r//3*3 + c//3
        for v in '123456789':
            if v in rows[r] or v in cols[c] or v in boxes[b]: continue
            board[r][c] = v; rows[r].add(v); cols[c].add(v); boxes[b].add(v)
            if bt(i + 1): return True
            board[r][c] = '.'; rows[r].discard(v); cols[c].discard(v); boxes[b].discard(v)
        return False
    bt(0)

def lcs(a, b):

    for i in range(m):
        for j in range(n):

def edit_distance(a, b):
    m, n = len(a), len(b); prev = list(range(n + 1))
    for i in range(1, m + 1):
        cur = [i] + [0] * n
        for j in range(1, n + 1):
            if a[i-1] == b[j-1]: cur[j] = prev[j-1]
            else: cur[j] = 1 + min(prev[j], cur[j-1], prev[j-1])   # delete, insert, replace
        prev = cur
    return prev[n]

def is_interleave(a, b, c):

def regex_match(s, p):                             # '.' and '*'

def distinct_subsequences(s, t):                   # ways t appears as subsequence of s

def insert(intervals, new):                        # intervals sorted, non-overlapping
    out = []; i = 0; n = len(intervals); s, e = new
    while i < n and intervals[i][1] < s: out.append(intervals[i]); i += 1

    out.append([s, e])
    return out + intervals[i:]

def rabin_karp(text, p, base=256, mod=10**9 + 7):
    n, m = len(text), len(p)
    if m > n: return []
    hp = ht = 0; high = pow(base, m - 1, mod); out = []
    for i in range(m):
        hp = (hp * base + ord(p[i])) % mod; ht = (ht * base + ord(text[i])) % mod
    for i in range(n - m + 1):
        if hp == ht and text[i:i + m] == p: out.append(i)
        if i + m < n: ht = ((ht - ord(text[i]) * high) * base + ord(text[i + m])) % mod
    return out
